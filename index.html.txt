function setView(v){ view=v; render(); }
function changeDate(delta){
  const d = new Date(currentDate);
  if(view==='daily') d.setDate(d.getDate()+delta);
  else if(view==='weekly') d.setDate(d.getDate()+delta*7);
  else if(view==='monthly') d.setMonth(d.getMonth()+delta);
  else if(view==='yearly') d.setFullYear(d.getFullYear()+delta);
  currentDate = d; render();
}
function getRange(){
  let start=new Date(currentDate), end=new Date(currentDate);
  if(view==='weekly'){ start.setDate(currentDate.getDate()-currentDate.getDay()); end.setDate(start.getDate()+6); }
  else if(view==='monthly'){ start.setDate(1); end.setMonth(currentDate.getMonth()+1,0); }
  else if(view==='yearly'){ start=new Date(currentDate.getFullYear(),0,1); end=new Date(currentDate.getFullYear(),11,31); }
  const range=[]; for(let d=new Date(start); d<=end; d.setDate(d.getDate()+1)) range.push(new Date(d));
  return range;
}

function updateField(cat, field, value, isRange){

  if(field) data[cat][field] = isRange ? Number(value) : value;

}

  const box = document.getElementById('emoSuggest');
  if(!q.trim()){ box.innerHTML=''; return; }
  const all = Object.values(EMOTION_CATEGORIES).flat();
  const cur = getEntry(currentDate).정신상태.emotions;
  const matches = all.filter(w=>w.includes(q) && !cur.includes(w)).slice(0,8);
  box.innerHTML = matches.length ? `<div class="emo-suggest">${matches.map(w=>`<div onclick="addEmotion('${w}')">${w}</div>`).join('')}</div>` : '';
}
function addEmotion(w){
  const data = getEntry(currentDate);
  if(!data.정신상태.emotions.includes(w)) data.정신상태.emotions.push(w);
  saveEntry(currentDate, data);
}
function removeEmotion(w){
  const data = getEntry(currentDate);
  data.정신상태.emotions = data.정신상태.emotions.filter(x=>x!==w);
  saveEntry(currentDate, data);
}

function renderYearly(){
  const year = currentDate.getFullYear();
  const months = Array.from({length:12},(_,m)=>{
    const monthDates=[]; const d=new Date(year,m,1); while(d.getMonth()===m){ monthDates.push(new Date(d)); d.setDate(d.getDate()+1); }
    const bodyScores = monthDates.map(dd=>entries[keyOf(dd)] && entries[keyOf(dd)].건강상태 && entries[keyOf(dd)].건강상태.score).filter(Boolean);
    const emoScores = monthDates.map(dd=>entries[keyOf(dd)] && entries[keyOf(dd)].정신상태 && entries[keyOf(dd)].정신상태.score).filter(Boolean);
    const avg = arr => arr.length ? Math.round((arr.reduce((a,b)=>a+Number(b),0)/arr.length)*10)/10 : null;
    return { label:(m+1)+'월', body:avg(bodyScores), emo:avg(emoScores) };
  });
  document.getElementById('content').innerHTML = `<div class="card"><h3 style="margin-bottom:10px;">${year}년 월별 평균 점수</h3>${lineChartSVG(months.map(m=>m.label), months.map(m=>m.body), months.map(m=>m.emo))}</div>`;
}
function chartHTML(range){
  const labels = range.map(d=>fmt(d).slice(5));
  const body = range.map(d=>getEntry(d).건강상태.score);
  const emo = range.map(d=>getEntry(d).정신상태.score);
  return `<div class="card" style="margin-top:12px;"><h3 style="margin-bottom:10px;">몸 & 감정 점수 추이</h3>${lineChartSVG(labels, body, emo)}</div>`;
}
function lineChartSVG(labels, bodyArr, emoArr){
  const w=680, h=220, padL=30, padB=24, padT=10, padR=10;
  const n = labels.length;
  const x = i => padL + (n<=1?0:(i*(w-padL-padR)/(n-1)));
  const y = v => v==null ? null : (h-padB) - ((v-0)/10)*(h-padB-padT);
  function pathFor(arr){
    let d=''; let started=false;
    arr.forEach((v,i)=>{
      const yy=y(v);
      if(yy==null){ started=false; return; }
      d += (started?' L ':'M ') + x(i) + ' ' + yy;
      started=true;
    });
    return d;
  }
  const gridLines = [0,2,4,6,8,10].map(v=>`<line x1="${padL}" y1="${y(v)}" x2="${w-padR}" y2="${y(v)}" stroke="#F0F0F0"/><text x="2" y="${y(v)+4}" font-size="9" fill="#999">${v}</text>`).join('');
  const xLabels = labels.map((l,i)=> i%Math.ceil(n/8||1)===0 ? `<text x="${x(i)}" y="${h-6}" font-size="9" fill="#999" text-anchor="middle">${l}</text>` : '').join('');
  return `<svg viewBox="0 0 ${w} ${h}" width="100%" height="220">
    ${gridLines}${xLabels}
    <path d="${pathFor(bodyArr)}" fill="none" stroke="${BODY_COLOR}" stroke-width="2.5"/>
    <path d="${pathFor(emoArr)}" fill="none" stroke="${EMOTION_COLOR}" stroke-width="2.5"/>
  </svg>
  <div style="font-size:12px;margin-top:4px;"><span style="color:${BODY_COLOR};font-weight:700;">● 몸</span> &nbsp; <span style="color:${EMOTION_COLOR};font-weight:700;">● 감정</span></div>`;
}

function exportCSV(){
  const rows = [['날짜','몸점수','아픈부위','건강메모','감정점수','감정단어','정신메모','미디어','일기','꿈일기']];
  Object.keys(entries).sort().forEach(k=>{
    const d = Object.assign(emptyData(), entries[k], { 건강상태: Object.assign(emptyData().건강상태, entries[k].건강상태||{}) });
    const date = fmt(new Date(k));
    const painList = Object.keys(d.건강상태.pain).map(id=>ALL_PART_LABELS[id]).join('/');
    rows.push([date, d.건강상태.score, painList, d.건강상태.text, d.정신상태.score, d.정신상태.emotions.join('/'), d.정신상태.text, d.미디어기록.text, d.일기, d.꿈일기]);
  });
  const csv = rows.map(r=>r.map(v=>`"${String(v??'').replace(/"/g,'""')}"`).join(',')).join('\n');
  const blob = new Blob(['\uFEFF'+csv], {type:'text/csv;charset=utf-8;'});
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a'); a.href=url; a.download='personal_record.csv';
  document.body.appendChild(a); a.click(); document.body.removeChild(a); URL.revokeObjectURL(url);
}
